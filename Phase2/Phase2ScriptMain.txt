# 4. Security Groups (Updated for Nginx Architecture)

# Nginx SG: Open to the world (Replaces ALB SG)
resource "aws_security_group" "nginx_sg" {
  name        = "nginx-proxy-sg"
  vpc_id      = aws_vpc.hydra_vpc.id
  description = "Allow Public HTTP and SSH Access"

  ingress {
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }
  ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }
  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

# Private Nodes SG: Only allow traffic from Nginx or Internal
resource "aws_security_group" "private_sg" {
  name        = "hydra-private-node-sg"
  vpc_id      = aws_vpc.hydra_vpc.id
  description = "Internal Communication"

  # Allow SSH from Nginx Bastion
  ingress {
    from_port       = 22
    to_port         = 22
    protocol        = "tcp"
    security_groups = [aws_security_group.nginx_sg.id]
  }
  # Allow App Traffic (5000-5001) from Nginx
  ingress {
    from_port       = 5000
    to_port         = 5001
    protocol        = "tcp"
    security_groups = [aws_security_group.nginx_sg.id]
  }
  # Allow Internal Traffic (Kafka 9092, Zookeeper 2181) from SELF
  ingress {
    from_port = 0
    to_port   = 0
    protocol  = "-1"
    self      = true
  }
  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

# 6. Instances (The Phase 2 Architecture)

# Nginx Bastion / Proxy
resource "aws_instance" "nginx" {
  ami                  = "ami-0e2c8caa4b6378d8c" # Ubuntu 24.04 (us-east-1)
  instance_type        = "t3.micro"
  subnet_id            = aws_subnet.public.id
  key_name             = aws_key_pair.kp.key_name
  iam_instance_profile = "LabInstanceProfile" # REQUIRED for Learner Lab
  security_groups      = [aws_security_group.nginx_sg.id]

  # Inject the Private IPs into the Nginx config automatically
  user_data = templatefile("nginx.conf.tpl", {
    node1_ip = aws_instance.private_nodes[0].private_ip
    node2_ip = aws_instance.private_nodes[1].private_ip
    node3_ip = aws_instance.private_nodes[2].private_ip
  })

  tags = { Name = "Hydra-Nginx-Proxy" }
}
