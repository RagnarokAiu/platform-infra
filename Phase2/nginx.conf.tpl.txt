#!/bin/bash
# Update and Install Nginx
sudo apt-get update -y
sudo apt-get install nginx -y

# Create the Nginx Configuration
cat <<EOT > /etc/nginx/sites-available/default
server {
    listen 80;

    # Route for TTS Service (Member 2) -> Node 1 (Port 5000)
    location /api/tts {
        proxy_pass http://${node1_ip}:5000;
        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
    }

    # Route for Document Service (Member 5) -> Node 1 (Port 5001)
    # Note: Sharing Node 1 with TTS but on a different port
    location /api/documents {
        proxy_pass http://${node1_ip}:5001;
        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
    }

    # Route for STT Service (Member 3) -> Node 2 (Port 5000)
    location /api/stt {
        proxy_pass http://${node2_ip}:5000;
        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
    }

    # Route for Quiz Service (Member 6) -> Node 2 (Port 5001)
    # Note: Sharing Node 2 with STT but on a different port
    location /api/quiz {
        proxy_pass http://${node2_ip}:5001;
        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
    }

    # Route for Chat Service (Member 4) -> Node 3 (Port 5000)
    location /api/chat {
        proxy_pass http://${node3_ip}:5000;
        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
    }

    # Health Check
    location / {
        return 200 'Project Hydra Nginx is Live!';
    }
}
EOT

# Restart Nginx
sudo systemctl restart nginx
